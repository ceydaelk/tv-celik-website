```typescript
"use client";

import { useState } from "react";
import Link from "next/link";
import { Menu, MessageCircle } from "lucide-react";
import MegaMenu from "@/components/layout/MegaMenu";
import MobileMenu from "@/components/layout/MobileMenu";

const NAV_LINKS = [
  { label: "Ana Sayfa", href: "/" },
  { label: "Kurumsal", href: "/kurumsal" },
  { label: "Projeler", href: "/projeler" },
  { label: "İletişim", href: "/iletisim" },
] as const;

const WHATSAPP_HREF =
  "https://wa.me/90XXXXXXXXXX?text=Merhaba%2C%20bilgi%20almak%20istiyorum.";

export default function Navbar() {
  const [megaMenuOpen, setMegaMenuOpen] = useState(false);
  const [mobileMenuOpen, setMobileMenuOpen] = useState(false);

  return (
    <>
      <header className="sticky top-0 z-50 bg-[#0f172a] shadow-lg">
        <nav className="mx-auto flex h-16 max-w-7xl items-center justify-between px-6">
          {/* Logo slot */}
          <Link
            href="/"
            className="text-xl font-bold tracking-wide text-white focus:outline-2 focus:outline-[#2563eb]"
          >
            TV ÇELİK
          </Link>

          {/* Desktop nav links */}
          <div className="hidden items-center gap-1 md:flex">
            {NAV_LINKS.map((link) => (
              <Link
                key={link.href}
                href={link.href}
                className="rounded-md px-3 py-2 text-sm font-normal text-white/90 transition-colors hover:bg-white/10 hover:text-white focus:outline-2 focus:outline-[#2563eb]"
              >
                {link.label}
              </Link>
            ))}

            {/* Hizmetler — triggers mega-menu on hover */}
            <div
              className="relative"
              onMouseEnter={() => setMegaMenuOpen(true)}
              onMouseLeave={() => setMegaMenuOpen(false)}
            >
              <button
                className="rounded-md px-3 py-2 text-sm font-normal text-white/90 transition-colors hover:bg-white/10 hover:text-white focus:outline-2 focus:outline-[#2563eb]"
                aria-haspopup="true"
                aria-expanded={megaMenuOpen}
              >
                Hizmetler
              </button>
              {megaMenuOpen && (
                <MegaMenu
                  onMouseEnter={() => setMegaMenuOpen(true)}
                  onMouseLeave={() => setMegaMenuOpen(false)}
                />
              )}
            </div>
          </div>

          {/* Right side: WhatsApp button (desktop) + Hamburger (mobile) */}
          <div className="flex items-center gap-3">
            {/* WhatsApp nav button — desktop only (D-03: FAB handles mobile) */}
            <a
              href={WHATSAPP_HREF}
              target="_blank"
              rel="noopener noreferrer"
              className="hidden items-center gap-2 rounded-full bg-[#25D366] px-5 py-2 text-sm font-bold text-white transition-colors duration-150 hover:bg-[#1da851] active:scale-95 focus:outline-2 focus:outline-white md:flex"
            >
              <MessageCircle size={16} />
              WhatsApp&apos;tan Yazın
            </a>

            {/* Hamburger — mobile only */}
            <button
              onClick={() => setMobileMenuOpen(true)}
              aria-label="Menüyü Aç"
              className="flex h-12 w-12 items-center justify-center text-white focus:outline-2 focus:outline-[#2563eb] md:hidden"
            >
              <Menu size={24} />
            </button>
          </div>
        </nav>
      </header>

      {/* MobileMenu rendered outside header to avoid stacking context issues */}
      <MobileMenu
        isOpen={mobileMenuOpen}
        onClose={() => setMobileMenuOpen(false)}
      />
    </>
  );
}
```
